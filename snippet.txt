  const id=await reqP(db.transaction(STORE,'readwrite').objectStore(STORE).add(item));
  const statusEl=document.getElementById('uploadStatus');
  try{
    // Prefer OPFS: stream the selected file directly instead of creating one huge Blob in memory.
    if(navigator.storage?.getDirectory){
      let path='reze_'+id+'_'+Date.now()+'_'+Math.random().toString(36).slice(2);
      try{
        const root=await navigator.storage.getDirectory();
        const fh=await root.getFileHandle(path,{create:true});
        const writable=await fh.createWritable();
        const reader=file.stream().getReader();let written=0;
        try{while(true){const {done,value}=await reader.read();if(done)break;await writable.write(value);written+=value.byteLength;if(statusEl)statusEl.textContent=`Đang thêm ${file.name} · ${Math.round(written/file.size*100)}%`}}finally{reader.releaseLock()}
        await writable.close();
        item.opfsPath=path;item.storage='opfs';delete item.chunked;delete item.chunkCount;await saveMeta({...item,id});return id;
      }catch(opfsErr){try{const root=await navigator.storage.getDirectory();await root.removeEntry(path)}catch(_){}console.warn('OPFS failed, falling back to chunks',opfsErr)}
    }
    item.chunked=true;item.chunkCount=Math.max(1,Math.ceil(file.size/CHUNK_SIZE));item.storage='chunks';await saveMeta({...item,id});
    for(let i=0;i<item.chunkCount;i++){const buf=await file.slice(i*CHUNK_SIZE,Math.min(file.size,(i+1)*CHUNK_SIZE)).arrayBuffer();await chunkPut(id,i,buf);if(statusEl)statusEl.textContent=`Đang thêm ${file.name} · ${Math.round((i+1)/item.chunkCount*100)}%`}
    return id;
  }catch(e){
    try{if(item.chunked)await deleteChunks(id)}catch(_){}
    try{if(item.opfsPath&&navigator.storage?.getDirectory){const root=await navigator.storage.getDirectory();await root.removeEntry(item.opfsPath)}}catch(_){}
    try{await reqP(db.transaction(STORE,'readwrite').objectStore(STORE).delete(id))}catch(_){}
    throw e;
  }
}
